function processSpelling(text) { const cleaned = text .trim() .replace(/[.!?]+$/g, "") .trim(); /* SPELL + WORD Example: "spell specific" Result: S-P-E-C-I-F-I-C */ const directWord = extractSpellCommand(cleaned); if (directWord) { spellingMode = false; waitingForSpellingWord = false; return spellWord(directWord); } /* SPELL BY ITSELF Example: "spell" The next single word will be spelled. */ if (isSpellCommand(cleaned)) { spellingMode = true; waitingForSpellingWord = true; return ""; } /* SPELL MODE If "spell" was spoken previously, the next single word is spelled. */ if ( spellingMode && waitingForSpellingWord ) { const words = cleaned.split(/\s+/); if (words.length === 1) { const word = words[0]; if (/^[A-Za-z]+$/.test(word)) { spellingMode = false; waitingForSpellingWord = false; return spellWord(word); } } /* The next speech was not a single word. Cancel spelling mode and allow normal transcription to handle it. */ spellingMode = false; waitingForSpellingWord = false; } return null; }
