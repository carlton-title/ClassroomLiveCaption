async function translateToThai(text) { try { const url = "https://api.mymemory.translated.net/get" + "?q=" + encodeURIComponent(text) + "&langpair=en|th"; const response = await fetch(url); if (!response.ok) { throw new Error( "Translation HTTP " + response.status ); } const data = await response.json(); if ( data.responseData && data.responseData.translatedText ) { return data.responseData.translatedText; } return null; } catch (error) { console.error( "Thai translation error:", error ); return null; } }
