function processSpelling(text) { const cleaned = text .trim() .replace(/[.!?]+$/g, "") .trim(); /* SPELL + WORD Example: "spell specific" → S-P-E-C-I-F-I-C */ const directWord = extractSpellCommand(cleaned); if (directWord) { spellingMode = false; waitingForSpellingWord = false; return spellWord(directWord); } /* SPELL BY ITSELF Example: "spell" Then the next word is spelled. */ if (isSpellCommand(cleaned)) { spellingMode = true; waitingForSpellingWord = true; return ""; } /* If SPELL mode is waiting for a word, spell the next single word. */ if ( spellingMode && waitingForSpellingWord ) { const words = cleaned.split(/\s+/); if (words.length === 1) { const word = words[0]; if (/^[A-Za-z]+$/.test(word)) { spellingMode = false; waitingForSpellingWord = false; return spellWord(word); } } /* If the next speech is not a single word, cancel spelling mode and let normal transcription handle it. */ spellingMode = false; waitingForSpellingWord = false; } return null; }
